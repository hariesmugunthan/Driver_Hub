class DriverModel {
  String driverId;
  String name;
  String mobileNumber;
  String email;
  DateTime? dateOfBirth;
  String? gender;
  String? address;
  String licenseNumber;
  DateTime licenseExpiryDate;
  bool isAvailable;

  DriverModel({
    this.driverId = '',
    required this.name,
    required this.mobileNumber,
    required this.email,
    this.dateOfBirth,
    this.gender,
    this.address,
    required this.licenseNumber,
    required this.licenseExpiryDate,
    this.isAvailable = true,
  });

  Map<String, dynamic> toJson() {
    return {
      'driver_id': driverId,
      'name': name,
      'mobile_number': mobileNumber,
      'email': email,
      'date_of_birth': dateOfBirth?.toIso8601String(),
      'gender': gender,
      'address': address,
      'license_number': licenseNumber,
      'license_expiry_date':
      licenseExpiryDate.toIso8601String(),
      'is_available': isAvailable,
    };
  }

  factory DriverModel.fromJson(
      Map<String, dynamic> json) {
    return DriverModel(
      driverId: json['driver_id'] as String? ?? '',
      name: json['name'] as String? ?? '',
      mobileNumber:
      json['mobile_number'] as String? ?? '',
      email: json['email'] as String? ?? '',
      dateOfBirth: json['date_of_birth'] != null
          ? DateTime.parse(
        json['date_of_birth'] as String,
      )
          : null,
      gender: json['gender'] as String?,
      address: json['address'] as String?,
      licenseNumber:
      json['license_number'] as String? ?? '',
      licenseExpiryDate: DateTime.parse(
        json['license_expiry_date'] as String,
      ),
      isAvailable:
      json['is_available'] as bool? ?? true,
    );
  }
}

import 'package:firebase_auth/firebase_auth.dart';
import 'package:google_sign_in/google_sign_in.dart';

class GoogleAuthService {
  Future<void> initializeGoogleSignIn() async {
    await GoogleSignIn.instance.initialize();
  }

  Future<String> signInWithGoogle() async {
    GoogleSignInAccount googleAccount =
    await GoogleSignIn.instance.authenticate();

    GoogleSignInAuthentication googleAuthentication =
        googleAccount.authentication;

    String? idToken = googleAuthentication.idToken;

    if (idToken == null) {
      throw Exception('Google ID token was not received');
    }

    OAuthCredential googleCredential = GoogleAuthProvider.credential(
      idToken: idToken,
    );

    await FirebaseAuth.instance.signInWithCredential(
      googleCredential,
    );

    String? firebaseIdToken =
    await FirebaseAuth.instance.currentUser?.getIdToken();

    if (firebaseIdToken == null) {
      throw Exception('Firebase ID token was not received');
    }

    return firebaseIdToken;
  }

  Future<void> signOutFromGoogle() async {
    await GoogleSignIn.instance.signOut();
    await FirebaseAuth.instance.signOut();
  }
}

import 'package:flutter/material.dart';

import '../../models/admin_registration_model.dart';

class AdminRegistrationDialog extends StatefulWidget {
  Future<void> Function(
      AdminRegistrationModel adminRegistrationModel,
      ) handleRegistration;

  AdminRegistrationDialog({
    super.key,
    required this.handleRegistration,
  });

  @override
  State<AdminRegistrationDialog> createState() =>
      AdminRegistrationDialogState();
}

class AdminRegistrationDialogState
    extends State<AdminRegistrationDialog> {
  TextEditingController nameController =
  TextEditingController();

  TextEditingController emailController =
  TextEditingController();

  TextEditingController masterPasswordController =
  TextEditingController();

  bool isRegistering = false;

  @override
  void dispose() {
    nameController.dispose();
    emailController.dispose();
    masterPasswordController.dispose();

    super.dispose();
  }

  Future<void> submitRegistration() async {
    String name = nameController.text.trim();
    String email = emailController.text.trim();
    String masterPassword =
        masterPasswordController.text;

    if (name.isEmpty ||
        email.isEmpty ||
        masterPassword.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('All fields are required'),
        ),
      );
      return;
    }

    AdminRegistrationModel adminRegistrationModel =
    AdminRegistrationModel(
      name: name,
      email: email,
      masterPassword: masterPassword,
    );

    setState(() {
      isRegistering = true;
    });

    try {
      await widget.handleRegistration(
        adminRegistrationModel,
      );

      if (!mounted) {
        return;
      }

      Navigator.pop(context);

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Admin registered successfully',
          ),
        ),
      );
    } catch (error) {
      if (!mounted) {
        return;
      }

      setState(() {
        isRegistering = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Admin registration failed',
          ),
        ),
      );
    }
  }

  Widget buildNameField() {
    return TextField(
      controller: nameController,
      decoration: const InputDecoration(
        labelText: 'Name',
        border: OutlineInputBorder(),
      ),
    );
  }

  Widget buildEmailField() {
    return TextField(
      controller: emailController,
      keyboardType: TextInputType.emailAddress,
      decoration: const InputDecoration(
        labelText: 'Email',
        border: OutlineInputBorder(),
      ),
    );
  }

  Widget buildMasterPasswordField() {
    return TextField(
      controller: masterPasswordController,
      obscureText: true,
      decoration: const InputDecoration(
        labelText: 'Master Password',
        border: OutlineInputBorder(),
      ),
    );
  }

  Widget buildForm() {
    return SingleChildScrollView(
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          buildNameField(),
          const SizedBox(height: 16),
          buildEmailField(),
          const SizedBox(height: 16),
          buildMasterPasswordField(),
        ],
      ),
    );
  }

  List<Widget> buildDialogActions() {
    return [
      TextButton(
        onPressed: isRegistering
            ? null
            : () {
          Navigator.pop(context);
        },
        child: const Text('Cancel'),
      ),
      ElevatedButton(
        onPressed:
        isRegistering ? null : submitRegistration,
        child: isRegistering
            ? const SizedBox(
          width: 20,
          height: 20,
          child: CircularProgressIndicator(),
        )
            : const Text('Register'),
      ),
    ];
  }

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      title: const Text('Register Admin'),
      content: buildForm(),
      actions: buildDialogActions(),
    );
  }
}
